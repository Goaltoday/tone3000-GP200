cmake_minimum_required(VERSION 3.22)
project(GP200ProtocolTests LANGUAGES CXX)
enable_testing()
add_executable(gp200_frames preset_frames_test.cpp)
target_compile_features(gp200_frames PRIVATE cxx_std_20)
# assert-based checks must stay enabled in Release too.
if(MSVC)
 target_compile_options(gp200_frames PRIVATE /UNDEBUG)
else()
 target_compile_options(gp200_frames PRIVATE -UNDEBUG -Wall -Wextra -Werror)
endif()
add_test(NAME gp200_frames COMMAND gp200_frames)

add_executable(gp200_codec codec_test.cpp ../../plugin/ui/views/gp200/GP200NAM.cpp)
target_include_directories(gp200_codec PRIVATE support)
target_compile_features(gp200_codec PRIVATE cxx_std_20)
if(MSVC)
 target_compile_options(gp200_codec PRIVATE /UNDEBUG)
else()
 target_compile_options(gp200_codec PRIVATE -UNDEBUG)
endif()
add_test(NAME gp200_codec COMMAND gp200_codec "${CMAKE_CURRENT_SOURCE_DIR}/compatible_fixture.nam")
